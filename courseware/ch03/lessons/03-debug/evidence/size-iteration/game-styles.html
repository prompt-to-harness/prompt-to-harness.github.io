/* ---------- 7 场地容器（居中 + 按玩法选场地容器） ---------- */

/* 7a. .board-wrap —— 只负责「居中」，与场地类型无关。
   所有玩法的场地都放在它里面；它不施加任何宽度约束。 */
.board-wrap { display: flex; justify-content: center; }

/* 7b. .stage —— 通用（非网格）场地容器：画布、自由布局、列表类玩法用这个。
   宽度上限由 --stage-max 控制，默认占满内容区；想更窄就在元素上写 --stage-max。
   内容纵向排列并水平居中，场地内部的提示文字用 .stage__tip。
   需要撑满宽度的子元素加 align-self: stretch。 */
.stage {
  display: flex;
  flex-direction: column;
  align-items: center;
  gap: clamp(8px, 1.6vw, 14px);
  width: 100%;
  max-width: var(--stage-max, 100%);
  min-width: 0;
}

.stage__tip {
  margin: 0;
  text-align: center;
  color: var(--text-dim);
  font-size: 0.85rem;
}

.stage__tip[hidden] { display: none; }

/* 7c. .board —— **网格场地容器**，仅供网格 / 棋盘类玩法
   （翻牌配对、消除、扫雷、2048 …）。
   ⚠️ .board 自带 display:grid 和「按列数算出的宽度上限」，
      它不是通用容器：把它挂到画布 / 自由布局的场地上，内容会被挤进第一列。
      非网格玩法请改用 .stage（见 7b）。 */
.board {
  --cols: 4;
  --rows: 4;
  --cell: 108px; /* 每列的宽度基准，按玩法需要调 */
  display: grid;
  grid-template-columns: repeat(var(--cols), minmax(0, 1fr));
  gap: clamp(6px, 1.4vw, 12px);
  width: 100%;
  max-width: min(100%, calc(var(--cols) * var(--cell)));
  perspective: 1400px;
}

/* 单块场地（棋盘类玩法）：
   <div class="board" style="--cols:4; --rows:4">…</div>
   .board__cell { aspect-ratio: 1/1; } */

/* 8c 轨道 / 目标（弹球、贪吃蛇、躲避收集等画布类玩法） */
/* 画布元素本身：宽度上限与长宽比都可调，不要把某一款游戏的尺寸写死。
   --stage-max   宽度上限（默认按 8 列 × 64px 估算，可按玩法覆盖）
   --stage-ratio 长宽比（默认 4/3，可按玩法改成 1/1、16/9 …） */
.stage-canvas {
  display: block;
  width: 100%;
  max-width: min(100%, var(--stage-max, calc(var(--cols, 8) * 64px)));
  aspect-ratio: var(--stage-ratio, 4 / 3);
  border-radius: var(--radius);
  border: 1px solid var(--border);

    radial-gradient(800px 400px at 50% 0%, rgba(124, 140, 255, 0.18), transparent 70%),
    rgba(0, 0, 0, 0.28);
  box-shadow: var(--shadow);
}
